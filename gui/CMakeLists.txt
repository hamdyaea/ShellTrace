cmake_minimum_required(VERSION 3.20)
project(ShellTraceGUI VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
qt_add_executable(shelltrace-gui main.cpp)
target_link_libraries(shelltrace-gui PRIVATE Qt6::Widgets)
include(GNUInstallDirs)
install(TARGETS shelltrace-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES shelltrace-gui.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES ../assets/shelltrace-badge.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME shelltrace.svg)
